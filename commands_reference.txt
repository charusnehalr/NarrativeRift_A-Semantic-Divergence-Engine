NarrativeRift — Commands Reference
====================================

Setup
------
# Create virtual environment
uv venv narrativerift

# Install dependencies
uv pip install -r requirements.txt --python narrativerift/Scripts/python.exe

# Activate virtual environment
narrativerift\Scripts\activate


Phase 2 — Chunking
-------------------
# Single article test
uv run python -m src.chunker --mode single --article aljazeera_001

# Batch with limit (use when daily quota is limited)
uv run python -m src.chunker --mode all --max 19

# Batch all articles (no limit)
uv run python -m src.chunker --mode all

# Different topic
uv run python -m src.chunker --mode all --max 19 --topic topic_b


Phase 3 — Embedding + ChromaDB
--------------------------------
# Embed a single article and store in ChromaDB
# Loads propositions JSON → embeds with BGE-M3 → upserts into ChromaDB
uv run python -m src.vector_store --mode single --article aljazeera_001

# Embed all chunked articles for a topic (skips already-embedded ones)
uv run python -m src.vector_store --mode all --topic topic_a

# Check stats — how many propositions are stored in ChromaDB for a topic
uv run python -m src.vector_store --mode stats --topic topic_a

# Semantic query — embed your query text and find the 5 most similar propositions
# Returns prop_id, cosine similarity score, text, source, and date
uv run python -m src.vector_store --mode query --topic topic_a --query "ceasefire negotiations"

# Verify Phase 3 is complete — checks two things:
#   1. metadata.json: all articles should show status=embedded
#   2. ChromaDB stats: total_propositions should be ~1500+ (30 articles x ~50 props each)
uv run python -m src.vector_store --mode stats --topic topic_a


Phase 4 — Contradiction Detection (NLI)
-----------------------------------------
# Run full contradiction detection pipeline for a topic
# DeBERTa scores all cross-doc pairs → all non-rejected pairs go to Groq for verification
# Output saved to data/topic_a/contradiction_pairs.json
uv run python -m src.nli_engine --topic topic_a

# Verify output — check how many contradiction pairs were found and how many used Groq
python -c "
import json
data = json.load(open('data/topic_a/contradiction_pairs.json'))
pairs = data['pairs']
tier2 = sum(1 for p in pairs if p.get('tier_2_used'))
print(f'Total pairs: {len(pairs)} | Groq verified: {tier2} | DeBERTa only: {len(pairs)-tier2}')
"

# Reset and re-run if you change thresholds (file is overwritten automatically)
uv run python -m src.nli_engine --topic topic_a


Phase 5 — Graph Builder
-------------------------
# Build contradiction graph from NLI pairs
# Loads contradiction_pairs.json → builds NetworkX graph → Louvain clustering
# Saves graph_data.json (for Streamlit) and graph.html (open in browser)
uv run python -m src.graph_builder --topic topic_a

# Open the interactive graph in browser (do NOT use VSCode preview)
# File location: data/topic_a/graph.html
# Right-click → Open With → Chrome or Edge

# Verify graph output
python -c "
import json
data = json.load(open('data/topic_a/graph_data.json'))
print(f'Nodes: {data[\"total_nodes\"]} | Edges: {data[\"total_edges\"]} | Communities: {data[\"total_communities\"]}')
"
